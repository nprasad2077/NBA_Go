  )

  // RateLimiter returns a per-IP rate limiter.
  // With 3 instances behind NGINX round-robin, effective limit ≈ Max * 3.
  func RateLimiter() fiber.Handler {
    return limiter.New(limiter.Config{
      Max:        20,              // requests per window per IP per instance
      Expiration: 1 * time.Minute, // sliding window
      KeyGenerator: func(c *fiber.Ctx) string {
        // Use X-Real-IP set by NGINX, fall back to remote IP
        if ip := c.Get("X-Real-IP"); ip != "" {
          return ip
        }
        return c.IP()
      },
      LimitReached: func(c *fiber.Ctx) error {
        return c.Status(fiber.StatusTooManyRequests).JSON(fiber.Map{
          "error": "Rate limit exceeded. Try again later.",
        })
      },
    })
  }

  package middleware

  import (
    "time"

    "github.com/gofiber/fiber/v2"
    "github.com/gofiber/fiber/v2/middleware/limiter"
  )

  // RateLimiter returns a per-IP rate limiter.
  // With 3 instances behind NGINX round-robin, effective limit ≈ Max * 3.
  func RateLimiter() fiber.Handler {
    return limiter.New(limiter.Config{
      Max:        20,              // requests per window per IP per instance
      Expiration: 1 * time.Minute, // sliding window
      KeyGenerator: func(c *fiber.Ctx) string {
        // Use X-Real-IP set by NGINX, fall back to remote IP
        if ip := c.Get("X-Real-IP"); ip != "" {
          return ip
        }
        return c.IP()
      },
      LimitReached: func(c *fiber.Ctx) error {
        return c.Status(fiber.StatusTooManyRequests).JSON(fiber.Map{
          "error": "Rate limit exceeded. Try again later.",
        })
      },
    })
  }

  package middleware

  import (
    "time"

    "github.com/gofiber/fiber/v2"
    "github.com/gofiber/fiber/v2/middleware/limiter"
  )

  // RateLimiter returns a per-IP rate limiter.
  // With 3 instances behind NGINX round-robin, effective limit ≈ Max * 3.
  func RateLimiter() fiber.Handler {
    return limiter.New(limiter.Config{
      Max:        20,              // requests per window per IP per instance
      Expiration: 1 * time.Minute, // sliding window
      KeyGenerator: func(c *fiber.Ctx) string {
        // Use X-Real-IP set by NGINX, fall back to remote IP
        if ip := c.Get("X-Real-IP"); ip != "" {
          return ip
        }
        return c.IP()
      },
      LimitReached: func(c *fiber.Ctx) error {
        return c.Status(fiber.StatusTooManyRequests).JSON(fiber.Map{
          "error": "Rate limit exceeded. Try again later.",
        })
      },
    })
  }
